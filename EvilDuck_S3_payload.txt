REM ===================================================
REM  EvilDuck_S3 - Ducky Script (educational demo)
REM  Открывает терминал (cmd), делает окно на весь экран (F12),
REM  меняет цвет текста на красный и выполняет команду dir
REM  Target: Windows
REM ===================================================

REM --- Пауза, чтобы устройство успело инициализироваться ---
DELAY 1000

REM --- Открываем окно "Выполнить" и запускаем командную строку ---
GUI r
DELAY 500
STRING cmd
ENTER
DELAY 1000

REM --- Разворачиваем терминал на полный экран (F12) ---
F12
DELAY 500

REM --- Меняем цвет текста на ярко-красный (0 = чёрный фон, C = красный текст) ---
STRING color 0C
ENTER
DELAY 300

REM --- Выводим список файлов текущего каталога ---
STRING dir
ENTER
